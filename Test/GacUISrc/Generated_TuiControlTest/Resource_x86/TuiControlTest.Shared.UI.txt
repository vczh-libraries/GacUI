            };
        }

        static func Install(<ls>locale : ::system::Locale, <ls>impl : ITuiStringResourceStrings^) : (void)
        {
            if (<ls>TuiStringResource.Keys.Contains(<ls>locale))
            {
                raise (("Localized strings \"tuidemo::TuiStringResource\" has already registered for locale \"" & <ls>locale) & "\".");
            }
            <ls>TuiStringResource.Set(<ls>locale, <ls>impl);
        }

        static func Get(<ls>locale : ::system::Locale) : (ITuiStringResourceStrings^)
        {
            if (<ls>TuiStringResource.Keys.Contains(<ls>locale))
            {
                return <ls>TuiStringResource[<ls>locale];
            }
            return <ls>TuiStringResource[(cast (::system::Locale) "en-US")];
        }

        new ()
        {
        }
    }
