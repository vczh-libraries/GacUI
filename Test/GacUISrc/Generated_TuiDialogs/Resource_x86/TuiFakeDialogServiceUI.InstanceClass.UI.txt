            if (<ls>TuiDialogStrings.Keys.Contains(<ls>locale))
            {
                raise (("Localized strings \"tui_controls::TuiDialogStrings\" has already registered for locale \"" & <ls>locale) & "\".");
            }
            <ls>TuiDialogStrings.Set(<ls>locale, <ls>impl);
        }

        static func Get(<ls>locale : ::system::Locale) : (ITuiDialogStringsStrings^)
        {
            if (<ls>TuiDialogStrings.Keys.Contains(<ls>locale))
            {
                return <ls>TuiDialogStrings[<ls>locale];
            }
            return <ls>TuiDialogStrings[(cast (::system::Locale) "en-US")];
        }

        new ()
        {
        }
    }
